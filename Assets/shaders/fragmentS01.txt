#version 120

uniform float time; // This matches your 'loc' binding variable

void main() {
    // Standardize UV coordinates using the raw texture space instead of window resolution
    vec2 uv = gl_TexCoord[0].xy * 2.0 - 1.0;

    // Calculate distance from center
    float d = length(uv);

    // Create the pulsing loop using your time uniform
    float pulse = 0.4 + 0.1 * sin(time * 3.0);
    float r = 0.02 / abs(d - pulse);

    // Calculate dynamic shifting colors
    vec3 color = vec3(0.5 + 0.5 * sin(time), 0.3, 0.5 + 0.5 * cos(time));
    color *= r;

    gl_FragColor = vec4(color, 1.0);
}
